import { createStorage } from "typebase-io/internal";
import { createLocalStorage } from "typebase-io/internal/local-storage";

export const localFileStorage = createLocalStorage({ root: "/cache/storage", url: "http://127.0.0.1:8080/storage" });

export const storage = createStorage({
  provider: "vercel",
  buckets: {
    avatars: { access: "public" },
    documents: { access: "private", prefix: "docs" },
  },
}, { local: localFileStorage });

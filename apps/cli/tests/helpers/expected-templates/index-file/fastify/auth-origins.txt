import Fastify from "fastify";
import cors from "@fastify/cors";
import { typebaseHandler } from "./server.ts";

const fastify = Fastify();

fastify.addContentTypeParser("*", (_request, _payload, done) => {
  done(null, undefined);
});

fastify.register(cors, {
  origin: ["https://app.com", "https://admin.com"],
  methods: ["GET", "POST", "PUT", "DELETE", "OPTIONS"],
  allowedHeaders: ["Content-Type", "Authorization"],
  credentials: true,
});

fastify.register(typebaseHandler);

fastify
  .listen({ port: 6000 })
  .then(() => console.log("Listening on 127.0.0.1:6000"));

import type { IncomingMessage, ServerResponse } from "node:http";
import { RPCHandler } from "@orpc/server/node";
import { CORSPlugin, RequestHeadersPlugin } from "@orpc/server/plugins";
import { onError } from "@orpc/server";

import { filterActions } from "typebase-io/internal";

import * as ActionModule0 from "./actions/files.ts";
import * as ActionModule1 from "./actions/mutations/todos.ts";
import * as ActionModule2 from "./actions/queries/todos.ts";

export const router = {
  files: filterActions(ActionModule0),
  mutations: {
    todos: filterActions(ActionModule1),
  },
  queries: {
    todos: filterActions(ActionModule2),
  },
};

const handler = new RPCHandler(router, {
  plugins: [
    new CORSPlugin({
      origin: (origin, options) => origin,
      allowMethods: ['GET', 'HEAD', 'PUT', 'POST', 'DELETE', 'PATCH'],
    }),
    new RequestHeadersPlugin(),
  ],
  interceptors: [
    onError((error) => {
      console.error(error);
    }),
  ],
});

export const typebaseHandler = async (req: IncomingMessage, res: ServerResponse, next?: (error?: unknown) => void) => {
  const { matched } = await handler.handle(req, res, {
    prefix: "/rpc",
    context: {},
  });

  if (matched) {
    return;
  }

  if (next) {
    next();
    return;
  }

  res.statusCode = 404;
  res.end("Not found");
};

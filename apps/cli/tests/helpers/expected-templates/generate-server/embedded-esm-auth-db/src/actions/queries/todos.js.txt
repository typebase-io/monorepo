import { ServerError } from "typebase-io/server";
import { z } from "zod";
import { authedAction } from "../custom-actions.js";
export const getOne = authedAction
    .input(z.object({
    id: z.number(),
}))
    .output(z.object({
    id: z.number(),
    value: z.string(),
    completed: z.boolean(),
}))
    .handler(async ({ db, input, user }) => {
    const todo = await db.query.todos.findFirst({
        where: { id: input.id, userId: user.id },
    });
    if (!todo) {
        throw new ServerError("NOT_FOUND");
    }
    return {
        id: todo.id,
        value: todo.value,
        completed: todo.completed,
    };
});
export const getMany = authedAction
    .output(z
    .object({
    id: z.number(),
    value: z.string(),
    completed: z.boolean(),
})
    .array())
    .handler(async ({ db, user }) => {
    const todos = await db.query.todos.findMany({
        where: { userId: user.id },
        orderBy: { createdAt: "desc" },
    });
    return todos.map((todo) => ({
        id: todo.id,
        value: todo.value,
        completed: todo.completed,
    }));
});

import { createStorage } from "typebase-io/internal";
import { createLocalStorage } from "typebase-io/internal/local-storage";

export const localFileStorage = createLocalStorage({ root: ".local-storage", url: "/files" });

export const storage = createStorage({
  provider: "vercel",
  buckets: { avatars: { access: "public" } },
}, { local: localFileStorage });

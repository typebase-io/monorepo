import type { IncomingMessage, ServerResponse } from "node:http";
import { RPCHandler } from "@orpc/server/node";
import { RequestHeadersPlugin } from "@orpc/server/plugins";
import { onError } from "@orpc/server";
import { handleNodeRequest } from "typebase-io/internal/local-storage";
import { localFileStorage } from "./storage.js";

import { filterActions } from "typebase-io/internal";

import * as ActionModule0 from "./actions/mutations/todos.js";
import * as ActionModule1 from "./actions/queries/todos.js";

export const router = {
  mutations: {
    todos: filterActions(ActionModule0),
  },
  queries: {
    todos: filterActions(ActionModule1),
  },
};

const handler = new RPCHandler(router, {
  plugins: [
    new RequestHeadersPlugin(),
  ],
  interceptors: [
    onError((error) => {
      console.error(error);
    }),
  ],
});

export const typebaseHandler = async (req: IncomingMessage, res: ServerResponse, next?: (error?: unknown) => void) => {
  const pathname = new URL(req.url ?? "/", "http://localhost").pathname;

  if (pathname === "/files" || pathname.startsWith("/files" + "/")) {
    return handleNodeRequest(localFileStorage.handle, req, res);
  }

  const { matched } = await handler.handle(req, res, {
    prefix: "/rpc",
    context: {},
  });

  if (matched) {
    return;
  }

  if (next) {
    next();
    return;
  }

  res.statusCode = 404;
  res.end("Not found");
};

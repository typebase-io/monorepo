import { os } from "@orpc/server";
import { Action } from "typebase-io/server";
import type { RequestHeadersPluginContext } from "@orpc/server/plugins";
import { auth } from "../auth.ts";
import { env } from "../env.ts";
import { publisher } from "../publisher.ts";

const base = os.$context<RequestHeadersPluginContext>();

const providerMiddleware = base
  .$context<{ auth?: typeof auth; env?: typeof env; publisher?: typeof publisher }>()
  .middleware(async ({ context, next }) => {
    return next({
      context: {
        auth: context.auth ?? auth,
        env: context.env ?? env,
        publisher: context.publisher ?? publisher,
      },
    });
  });

const withProviders = base.use(providerMiddleware);

export const action = new Action(withProviders);
